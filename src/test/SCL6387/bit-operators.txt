/* SCL-6387 / SCL-6388: bitwise operators of the ABL Agent (BIT-AND, BIT-OR, BIT-XOR,
   BIT-NOT, BIT-NAND, BIT-NOR, BIT-XNOR, BIT-SHL, BIT-SHR, BIT-LSHR).
   Only the ABL Agent compiles this source (verified with ablagent run, the results are noted

   a .txt file and not a .p file. */

define variable iB      as integer no-undo initial 3 .
define variable iR      as int64   no-undo .
define variable lR      as logical no-undo .
define variable bit-or  as integer no-undo initial 1 .
define variable bit-not as integer no-undo .

/* every operator */

iR = iA bit-shr iB .                      /* 0 */

/* BIT-NOT with parentheses, nested, below an additive operator */
iR = bit-not (iA) .                       /* -7 */

iR = bit-not iA + 1 .                     /* -6 */

/* precedence levels */
iR = 2 * iA bit-or 1 .                    /* 13 */

iR = 4 bit-xor 4 bit-and 0 .              /* 4 */

lR = (iA bit-or iB) = 7 .                 /* yes */
lR = bit-not 1 = -2 .                     /* yes */
iR = - iA bit-and 3 .                     /* 2 */
iR = iA bit-and iB bit-xor iA bit-or iB . /* 7 */
